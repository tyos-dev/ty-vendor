cmake_minimum_required(VERSION 3.30)
project(ty-[[TY-LIB-NAME]]-wrapper)

include(CMakePackageConfigHelpers)

add_library([[TY-LIB-NAME]] INTERFACE)
add_library([[TY-LIB-NAME]]::[[TY-LIB-NAME]] ALIAS [[TY-LIB-NAME]])

target_include_directories([[TY-LIB-NAME]]
        INTERFACE
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/[[TY-LIB-NAME]]>
        $<INSTALL_INTERFACE:include>
)

configure_package_config_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/cmake/[[TY-LIB-NAME]]Config.cmake.in"
        "${CMAKE_CURRENT_BINARY_DIR}/[[TY-LIB-NAME]]Config.cmake"
        INSTALL_DESTINATION lib/cmake/[[TY-LIB-NAME]]
)

write_basic_package_version_file(
        "${CMAKE_CURRENT_BINARY_DIR}/[[TY-LIB-NAME]]ConfigVersion.cmake"
        VERSION 1.0.0
        COMPATIBILITY AnyNewerVersion
)

install(DIRECTORY
        [[TY-LIB-NAME]]
        DESTINATION include/
)

install(FILES
        "${CMAKE_CURRENT_BINARY_DIR}/[[TY-LIB-NAME]]Config.cmake"
        "${CMAKE_CURRENT_BINARY_DIR}/[[TY-LIB-NAME]]ConfigVersion.cmake"
        DESTINATION lib/cmake/[[TY-LIB-NAME]]
)

install(TARGETS [[TY-LIB-NAME]]
        EXPORT [[TY-LIB-NAME]]Targets
)

install(EXPORT [[TY-LIB-NAME]]Targets
        FILE [[TY-LIB-NAME]]Targets.cmake
        NAMESPACE [[TY-LIB-NAME]]::
        DESTINATION lib/cmake/[[TY-LIB-NAME]]
)
